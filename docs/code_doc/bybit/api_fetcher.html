119    async def get_funding_rates(self, klines_df: pd.DataFrame, product: str) -> pd.DataFrame:
120        """Associate the funding rate of a contract to each kline.
121
122        This function fetches all required funding rates by paginating through
123        Bybit's API until all klines are covered.
124
125        Args:
126            klines_df (pd.DataFrame): The DataFrame containing the klines.
127            product (str): The product to get the funding rates from.
128
129        Returns:
130            pd.DataFrame: The DataFrame with the funding rates.
131
132        """
133        end_time = klines_df["startTime"].max()  # Newest timestamp in klines
134        limit = klines_df["startTime"].min()  # Oldest timestamp in klines
135
136        # Start with the current fundingRate
137        current_funding = self.session.get_tickers(symbol=product, category="linear")["result"]["list"]
138
139        current_funding_df = pd.DataFrame(current_funding)
140        # Remove a 7 hours, 59 minutes and 50 seconds in epoch milliseconds
141        current_funding_df["fundingRateTimestamp"] = current_funding_df["nextFundingTime"].astype(int) - 28_790_000
142        current_funding_df["fundingRate"] = current_funding_df["fundingRate"].astype(float)
143        funding_data = [current_funding_df[["fundingRateTimestamp", "fundingRate"]]]
144
145        # If we fundingRateTimestamp column, add its data to the funding data
146        if "fundingRate" in klines_df.columns:
147            klines_df = klines_df.rename(columns={"fundingRate": "existingFundingRate"})
148
149        while True:
150            # Query funding rate history
151            params = {
152                "category": "linear",
153                "symbol": product,
154                "endTime": end_time,
155            }
156            response = self.session.get_funding_rate_history(**params)["result"]["list"]
157
158            self.logger.info(f"Fetched {len(response)} new funding rate data points.")
159
160            if not response:  # No more data
161                break
162
163            # Convert to DataFrame and append
164            df_funding = pd.DataFrame(response)[["fundingRateTimestamp", "fundingRate"]]
165            df_funding["fundingRateTimestamp"] = df_funding["fundingRateTimestamp"].astype(int)
166            df_funding["fundingRate"] = df_funding["fundingRate"].astype(float)
167            funding_data.append(df_funding)
168
169            # Update start_time for the next batch (use last timestamp)
170            end_time = df_funding["fundingRateTimestamp"].min()
171
172            # If last funding rate is beyond the max kline timestamp, stop
173            if end_time < limit:
174                break
175
176            await asyncio.sleep(0.1)
177
178        # Combine all funding data
179        if not funding_data:
180            msg = "No funding rate data found."
181            raise ValueError(msg)
182
183        df_funding_final = pd.concat(funding_data).sort_values("fundingRateTimestamp")
184
185        klines_df = klines_df.sort_values("startTime")
186        # Merge using nearest past funding rate timestamp
187        df_merged = (
188            pd.merge_asof(
189                klines_df, df_funding_final, left_on="startTime", right_on="fundingRateTimestamp", direction="backward"
190            )
191            .sort_values("startTime", ascending=False)
192            .reset_index(drop=True)
193            .drop(columns=["fundingRateTimestamp"])
194        )
195
196        # Combine with the existing funding rate column
197        if "existingFundingRate" in df_merged.columns:
198            df_merged["fundingRate"] = df_merged["fundingRate"].combine_first(df_merged["existingFundingRate"])
199            df_merged = df_merged.drop(columns=["existingFundingRate"])
200
201        return df_merged

205    async def get_history_pd(
206        self,
207        product: str,
208        interval: str = "m",
209        dateLimit: str = "01/01/2024",
210        category: str = "linear",
211        dest: str | None = None,
212    ) -> pd.DataFrame:
213        """Get the history of a future product until dateLimit.
214
215        If we do not have any data, we start from the oldest data point, and fetch the data before it
216        If we have some data, we start from the most recent data point, and fetch the data after it
217        We do it this way, because we cannot know when the product started
218        Also, when a product has no more klines, it will not throw an error
219
220        Warning: the last candle will not be at dateLimit, but a little after it
221
222        Link: https://bybit-exchange.github.io/docs/v5/market/kline
223        Args:
224            product (str): The future product to get the history from
225            interval (str): The interval of the data
226            dateLimit (str): The last date of fetched data
227            category (str): The category of the product
228            dest (str | None): The destination folder to save the data
229        Returns:
230            A DataFrame containing the accumulated data
231        """
232        file_name = f"{product}_{interval}"
233        if category == "spot":
234            file_name += "_spot.parquet"
235        else:
236            file_name += ".parquet"
237
238        if dest:
239            file_name = Path(dest) / file_name
240
241        dateLimit = get_epoch(dateLimit)
242
243        ORANGE = "\033[38;5;214m"
244        RESET = "\033[0m"
245        self.logger.info(f"Fetching data for {ORANGE}{product}{RESET} in {ORANGE}{interval}{RESET} interval.")
246        try:
247            acc_data = load_klines_parquet(file_name)
248            self.logger.info(f"Loaded {len(acc_data)} existing data points.")
249            timestamp_key = "start"
250            timestamp = acc_data.iloc[0]["startTime"]
251
252        except FileNotFoundError:
253            # Initialize an empty DataFrame for accumulated data
254            acc_data = pd.DataFrame(
255                columns=["startTime", "openPrice", "highPrice", "lowPrice", "closePrice", "volume", "turnover"],
256            )
257            self.logger.info("No previous data found, starting fresh.")
258            timestamp_key = "end"
259            timestamp = None
260
261        params = {
262            "symbol": product,
263            "category": category,
264            "interval": interval,
265            "limit": 1000,
266        }
267
268        while True:
269            if timestamp:
270                params[timestamp_key] = timestamp
271
272            response = self.session.get_kline(**params)["result"]["list"]
273            new_data = pd.DataFrame(
274                response,
275                columns=["startTime", "openPrice", "highPrice", "lowPrice", "closePrice", "volume", "turnover"],
276            )
277
278            self.logger.info(f"Fetched {len(new_data)} new data points.")
279
280            if timestamp_key == "start":
281                acc_data = pd.concat([new_data, acc_data.iloc[1:]], ignore_index=True)
282            else:
283                acc_data = pd.concat([acc_data.iloc[:-1], new_data], ignore_index=True)
284
285            numberCandles = len(new_data)
286            if numberCandles > 0:
287                timestamp = (
288                    acc_data.iloc[0]["startTime"] if timestamp_key == "start" else acc_data.iloc[-1]["startTime"]
289                )
290
291            if numberCandles < 1000 or int(acc_data.iloc[-1]["startTime"]) < dateLimit:
292                break
293
294        # Affect types
295        acc_data = acc_data.astype(
296            {
297                "startTime": "int",
298                "openPrice": "float",
299                "highPrice": "float",
300                "lowPrice": "float",
301                "closePrice": "float",
302                "volume": "float",
303                "turnover": "float",
304            }
305        )
306
307        if product in PERPETUALS:
308            acc_data = await self.get_funding_rates(klines_df=acc_data, product=product)
309
310        if not acc_data.empty:
311            save_klines_parquet(file_name, acc_data)
312        return acc_data

317        self,
318        coin: str = "BTC",
319        datelimit: str = "01/01/2024",
320        dest: str = "store",
321        spot: bool = True,
322        perpetual: bool = True,
323        inverse: bool = True,
324    ) -> None:
325        """Save the klines of all the Perpetual/Future/Inverse contracts in parquet format.
326
327        Checks if a parquet file exists to update it, else creates a new one
328
329        Args:
330            coin (str): The coin to consider (e.g., "BTC").
331            datelimit (str): The last date of fetched data.
332            dest (str): The destination folder to save the data.
333            spot (bool): Include spot contracts.
334            perpetual (bool): Include perpetual contracts.
335            inverse (bool): Include inverse contracts.
336
337        """
338        allContracts = self.get_linearNames(inverse=inverse, perpetual=perpetual, coin=coin)
339
340        # Combine perpetual and future contracts
341        allContracts = allContracts["perpetual"] + allContracts["future"]
342
343        async def _fetch_history(contract: str, interval: str, category: str = "linear") -> None:
344            await self.get_history_pd(
345                product=contract, dateLimit=datelimit, interval=interval, dest=dest, category=category
346            )
347
348        tasks = []
349
350        for contract in allContracts:
351            tasks.extend(
352                [_fetch_history(contract, "15"), _fetch_history(contract, "5"), _fetch_history(contract, "1")],
353            )
354
355        # spot
356        if spot:
357            tasks.extend(
358                [
359                    _fetch_history(f"{coin}USDT", "15", category="spot"),
360                    _fetch_history(f"{coin}USDT", "5", category="spot"),
361                    _fetch_history(f"{coin}USDT", "1", category="spot"),
362                ],
363            )
364
365        # TODO: This code will wait for other tasks, because we use I/O bound tasks
366        await asyncio.gather(*tasks)

420    @beartype
421    def all_gaps_pd(
422        self,
423        coin: str = "BTC",
424        quoteCoins: list[str] = ["USDC", "USDT", "USD"],
425        spot: bool = False,
426        perpetual: bool = True,
427        inverse: bool = False,
428    ) -> pd.DataFrame:  # Function can return either DataFrame or Styler
429        """Get all the gaps for multiple products in a DataFrame.
430
431        WARNING: There is no Future/Inverse with USD or USDT, so selecting these will return nothing.
432
433        Args:
434            coin (str): The coin to consider (e.g., "BTC").
435            quoteCoins (list[str]): Quote currencies to consider (e.g., ["USDC", "USDT"]).
436            spot (bool): Include spot contracts.
437            perpetual (bool): Use perpetual contracts.
438            inverse (bool): Use inverse contracts.
439
440        Returns:
441            pd.DataFrame | pd.io.formats.style.Styler: A DataFrame or a styled version.
442
443        """
444        # Get future and spot contracts
445        market = self.get_linearNames(coin=coin, inverse=inverse, perpetual=perpetual, quoteCoins=quoteCoins)
446
447        shortTickers = [
448            self.session.get_tickers(symbol=future, category="linear")["result"] for future in market["future"]
449        ]
450
451        longTickers = [
452            self.session.get_tickers(symbol=perpetual, category="linear")["result"] for perpetual in market["perpetual"]
453        ]
454
455        if spot:
456            for stableCoin in ["USDT", "USDC"]:
457                if stableCoin in quoteCoins:
458                    response = self.session.get_tickers(symbol=f"{coin}{stableCoin}", category="spot")["result"]
459                    response["list"][0]["deliveryTime"] = 0  # Spot contracts don't have a delivery time
460                    response["list"][0]["symbol"] = f"{coin}{stableCoin} (Spot)"
461                    longTickers.append(response)
462
463        # Define the column types
464        column_types = {
465            "Buy": "string",
466            "Sell": "string",
467            "Gap": "float",
468            "Coeff": "float",
469            "ROI": "float",
470            "APR": "float",
471            "CumFundingRate": "float",
472            "CumVolume": "int",
473            "DaysLeft": "int",
474        }
475
476        df_gaps = pd.DataFrame(columns=column_types.keys()).astype(column_types)
477
478        # Calculate gaps
479        for _i, long in enumerate(longTickers):
480            for short in shortTickers:
481                # Cross only the products in different categories
482                longInfo = long["list"][0]
483                shortInfo = short["list"][0]
484                gap = Analyser.get_gap(longInfo, shortInfo)
485
486                row = pd.DataFrame(
487                    [
488                        {
489                            "Buy": longInfo["symbol"],
490                            "Sell": shortInfo["symbol"],
491                            "Gap": gap["gap"],
492                            "Coeff": gap["coeff"],
493                            "ROI": gap["roi"],
494                            "APR": gap["apr"],
495                            "CumFundingRate": gap["cumFunding"],
496                            "CumVolume": gap["cumVolume"],
497                            "DaysLeft": max(0, int(gap["daysLeft"])),
498                        },
499                    ],
500                )
501
502                df_gaps = pd.concat([df_gaps, row], ignore_index=True)
503
504        df_gaps = df_gaps.astype(column_types)
505
506        # Sort by DaysLeft
507        df_gaps = df_gaps.sort_values(by="DaysLeft")
508
509        return df_gaps.reset_index(drop=True)

120    async def get_funding_rates(self, klines_df: pd.DataFrame, product: str) -> pd.DataFrame:
121        """Associate the funding rate of a contract to each kline.
122
123        This function fetches all required funding rates by paginating through
124        Bybit's API until all klines are covered.
125
126        Args:
127            klines_df (pd.DataFrame): The DataFrame containing the klines.
128            product (str): The product to get the funding rates from.
129
130        Returns:
131            pd.DataFrame: The DataFrame with the funding rates.
132
133        """
134        end_time = klines_df["startTime"].max()  # Newest timestamp in klines
135        limit = klines_df["startTime"].min()  # Oldest timestamp in klines
136
137        # Start with the current fundingRate
138        current_funding = self.session.get_tickers(symbol=product, category="linear")["result"]["list"]
139
140        current_funding_df = pd.DataFrame(current_funding)
141        # Remove a 7 hours, 59 minutes and 50 seconds in epoch milliseconds
142        current_funding_df["fundingRateTimestamp"] = current_funding_df["nextFundingTime"].astype(int) - 28_790_000
143        current_funding_df["fundingRate"] = current_funding_df["fundingRate"].astype(float)
144        funding_data = [current_funding_df[["fundingRateTimestamp", "fundingRate"]]]
145
146        # If we fundingRateTimestamp column, add its data to the funding data
147        if "fundingRate" in klines_df.columns:
148            klines_df = klines_df.rename(columns={"fundingRate": "existingFundingRate"})
149
150        while True:
151            # Query funding rate history
152            params = {
153                "category": "linear",
154                "symbol": product,
155                "endTime": end_time,
156            }
157            response = self.session.get_funding_rate_history(**params)["result"]["list"]
158
159            self.logger.info(f"Fetched {len(response)} new funding rate data points.")
160
161            if not response:  # No more data
162                break
163
164            # Convert to DataFrame and append
165            df_funding = pd.DataFrame(response)[["fundingRateTimestamp", "fundingRate"]]
166            df_funding["fundingRateTimestamp"] = df_funding["fundingRateTimestamp"].astype(int)
167            df_funding["fundingRate"] = df_funding["fundingRate"].astype(float)
168            funding_data.append(df_funding)
169
170            # Update start_time for the next batch (use last timestamp)
171            end_time = df_funding["fundingRateTimestamp"].min()
172
173            # If last funding rate is beyond the max kline timestamp, stop
174            if end_time < limit:
175                break
176
177            await asyncio.sleep(0.1)
178
179        # Combine all funding data
180        if not funding_data:
181            msg = "No funding rate data found."
182            raise ValueError(msg)
183
184        df_funding_final = pd.concat(funding_data).sort_values("fundingRateTimestamp")
185
186        klines_df = klines_df.sort_values("startTime")
187        # Merge using nearest past funding rate timestamp
188        df_merged = (
189            pd.merge_asof(
190                klines_df, df_funding_final, left_on="startTime", right_on="fundingRateTimestamp", direction="backward"
191            )
192            .sort_values("startTime", ascending=False)
193            .reset_index(drop=True)
194            .drop(columns=["fundingRateTimestamp"])
195        )
196
197        # Combine with the existing funding rate column
198        if "existingFundingRate" in df_merged.columns:
199            df_merged["fundingRate"] = df_merged["fundingRate"].combine_first(df_merged["existingFundingRate"])
200            df_merged = df_merged.drop(columns=["existingFundingRate"])
201
202        return df_merged

206    async def get_history_pd(
207        self,
208        product: str,
209        interval: str = "m",
210        dateLimit: str = "01/01/2024",
211        category: str = "linear",
212        dest: str | None = None,
213    ) -> pd.DataFrame:
214        """Get the history of a future product until dateLimit.
215
216        If we do not have any data, we start from the oldest data point, and fetch the data before it
217        If we have some data, we start from the most recent data point, and fetch the data after it
218        We do it this way, because we cannot know when the product started
219        Also, when a product has no more klines, it will not throw an error
220
221        Warning: the last candle will not be at dateLimit, but a little after it
222
223        Link: https://bybit-exchange.github.io/docs/v5/market/kline
224        Args:
225            product (str): The future product to get the history from
226            interval (str): The interval of the data
227            dateLimit (str): The last date of fetched data
228            category (str): The category of the product
229            dest (str | None): The destination folder to save the data
230        Returns:
231            A DataFrame containing the accumulated data
232        """
233        file_name = f"{product}_{interval}"
234        if category == "spot":
235            file_name += "_spot.parquet"
236        else:
237            file_name += ".parquet"
238
239        if dest:
240            file_name = Path(dest) / file_name
241
242        dateLimit = get_epoch(dateLimit)
243
244        ORANGE = "\033[38;5;214m"
245        RESET = "\033[0m"
246        self.logger.info(f"Fetching data for {ORANGE}{product}{RESET} in {ORANGE}{interval}{RESET} interval.")
247        try:
248            acc_data = load_klines_parquet(file_name)
249            self.logger.info(f"Loaded {len(acc_data)} existing data points.")
250            timestamp_key = "start"
251            timestamp = acc_data.iloc[0]["startTime"]
252
253        except FileNotFoundError:
254            # Initialize an empty DataFrame for accumulated data
255            acc_data = pd.DataFrame(
256                columns=["startTime", "openPrice", "highPrice", "lowPrice", "closePrice", "volume", "turnover"],
257            )
258            self.logger.info("No previous data found, starting fresh.")
259            timestamp_key = "end"
260            timestamp = None
261
262        params = {
263            "symbol": product,
264            "category": category,
265            "interval": interval,
266            "limit": 1000,
267        }
268
269        while True:
270            if timestamp:
271                params[timestamp_key] = timestamp
272
273            response = self.session.get_kline(**params)["result"]["list"]
274            new_data = pd.DataFrame(
275                response,
276                columns=["startTime", "openPrice", "highPrice", "lowPrice", "closePrice", "volume", "turnover"],
277            )
278
279            self.logger.info(f"Fetched {len(new_data)} new data points.")
280
281            if timestamp_key == "start":
282                acc_data = pd.concat([new_data, acc_data.iloc[1:]], ignore_index=True)
283            else:
284                acc_data = pd.concat([acc_data.iloc[:-1], new_data], ignore_index=True)
285
286            numberCandles = len(new_data)
287            if numberCandles > 0:
288                timestamp = (
289                    acc_data.iloc[0]["startTime"] if timestamp_key == "start" else acc_data.iloc[-1]["startTime"]
290                )
291
292            if numberCandles < 1000 or int(acc_data.iloc[-1]["startTime"]) < dateLimit:
293                break
294
295        # Affect types
296        acc_data = acc_data.astype(
297            {
298                "startTime": "int",
299                "openPrice": "float",
300                "highPrice": "float",
301                "lowPrice": "float",
302                "closePrice": "float",
303                "volume": "float",
304                "turnover": "float",
305            }
306        )
307
308        if product in PERPETUALS:
309            acc_data = await self.get_funding_rates(klines_df=acc_data, product=product)
310
311        if not acc_data.empty:
312            save_klines_parquet(file_name, acc_data)
313        return acc_data
